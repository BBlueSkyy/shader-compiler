cmake_minimum_required(VERSION 3.20)
project(storage_subword_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(HOST_DEPS "" CACHE PATH "Directory containing fmt and sirit source checkouts")
set(BOOST_INCLUDE_DIR "" CACHE PATH "Boost headers")
find_program(SPIRV_VAL spirv-val REQUIRED)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(SIRIT_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${HOST_DEPS}/fmt fmt EXCLUDE_FROM_ALL)
add_subdirectory(${HOST_DEPS}/sirit sirit EXCLUDE_FROM_ALL)
get_filename_component(DEFAULT_COMPILER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(COMPILER_ROOT "${DEFAULT_COMPILER_ROOT}" CACHE PATH "Shader compiler sources")
option(TEST_STORAGE_FEATURES "Exercise separate subword storage feature flags" ON)
# The production sources use shader_compiler/ includes even though the repository has a hyphen.
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include)
file(CREATE_LINK ${COMPILER_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/include/shader_compiler SYMBOLIC)
file(GLOB SPIRV_SOURCES CONFIGURE_DEPENDS ${COMPILER_ROOT}/backend/spirv/*.cpp)
file(GLOB IR_SOURCES CONFIGURE_DEPENDS ${COMPILER_ROOT}/frontend/ir/*.cpp)
list(FILTER IR_SOURCES EXCLUDE REGEX "/(post_order|ir_emitter)\\.cpp$")
# Both regression targets share the production backend objects: CI compiles once.
add_library(shader_backend_tests STATIC ${SPIRV_SOURCES} ${IR_SOURCES})
target_include_directories(shader_backend_tests PUBLIC ${CMAKE_CURRENT_BINARY_DIR}/include ${BOOST_INCLUDE_DIR} ${HOST_DEPS}/range/include)
target_link_libraries(shader_backend_tests PUBLIC fmt::fmt sirit)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(shader_backend_tests PRIVATE -Werror=conversion)
endif()
add_executable(storage_subword_tests generate.cpp)
target_link_libraries(storage_subword_tests PRIVATE shader_backend_tests)
if(TEST_STORAGE_FEATURES)
    target_compile_definitions(storage_subword_tests PRIVATE TEST_STORAGE_FEATURES)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME storage_subword COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/verify.py $<TARGET_FILE:storage_subword_tests> ${SPIRV_VAL} ${CMAKE_CURRENT_BINARY_DIR}/modules)

option(TEST_STORAGE_IMAGES "Build image capability regression suite with shared backend" ON)
option(TEST_TYPELESS_WRITES "Exercise typeless write capability gate" ON)
if(TEST_STORAGE_IMAGES)
    add_executable(storage_images_tests ../storage_images/generate.cpp)
    target_link_libraries(storage_images_tests PRIVATE shader_backend_tests)
    if(TEST_TYPELESS_WRITES)
        target_compile_definitions(storage_images_tests PRIVATE TEST_TYPELESS_WRITES)
    endif()
    add_test(NAME storage_images COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../storage_images/verify.py $<TARGET_FILE:storage_images_tests> ${SPIRV_VAL} ${CMAKE_CURRENT_BINARY_DIR}/image-modules)
endif()
