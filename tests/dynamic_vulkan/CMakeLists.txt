cmake_minimum_required(VERSION 3.20)
project(dynamic_vulkan_execution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
add_executable(dynamic_vulkan_runner runner.cpp)
target_link_libraries(dynamic_vulkan_runner PRIVATE Vulkan::Vulkan)
set(DYNAMIC_SAMPLE_SPV "" CACHE FILEPATH "Production dynamic_texture_tests sample.spv")
if(DYNAMIC_SAMPLE_SPV)
    enable_testing()
    add_test(NAME dynamic_vulkan_execution COMMAND dynamic_vulkan_runner "${DYNAMIC_SAMPLE_SPV}")
    set_tests_properties(dynamic_vulkan_execution PROPERTIES SKIP_RETURN_CODE 77)
    if(DYNAMIC_TEST_DEPENDENCY)
        set_tests_properties(dynamic_vulkan_execution PROPERTIES DEPENDS "${DYNAMIC_TEST_DEPENDENCY}")
    endif()
endif()
set(DYNAMIC_MODULE_DIR "" CACHE PATH "Directory containing all production dynamic_texture_tests modules")
if(DYNAMIC_MODULE_DIR)
    enable_testing()
    foreach(operation sample transform phi fetch query)
        if(operation STREQUAL "transform" OR operation STREQUAL "phi")
            set(expectation sample)
        else()
            set(expectation "${operation}")
        endif()
        foreach(suffix "" "_65536" "_66560")
            set(test_name "dynamic_vulkan_${operation}${suffix}")
            add_test(NAME "${test_name}" COMMAND dynamic_vulkan_runner
                "${DYNAMIC_MODULE_DIR}/${operation}${suffix}.spv" "${expectation}")
            set_tests_properties("${test_name}" PROPERTIES SKIP_RETURN_CODE 77)
            if(DYNAMIC_TEST_DEPENDENCY)
                set_tests_properties("${test_name}" PROPERTIES DEPENDS "${DYNAMIC_TEST_DEPENDENCY}")
            endif()
        endforeach()
    endforeach()
endif()
