cmake_minimum_required(VERSION 3.20)
project(handle_tracking_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(HOST_DEPS "" CACHE PATH "Host fmt, range source and Boost headers")
set(BOOST_INCLUDE_DIR "" CACHE PATH "Boost headers")
set(FMT_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(${HOST_DEPS}/fmt fmt EXCLUDE_FROM_ALL)
get_filename_component(COMPILER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include)
file(CREATE_LINK ${COMPILER_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/include/shader_compiler SYMBOLIC)
file(GLOB IR_SOURCES CONFIGURE_DEPENDS ${COMPILER_ROOT}/frontend/ir/*.cpp)
list(FILTER IR_SOURCES EXCLUDE REGEX "/post_order\\.cpp$")
add_executable(handle_tracking_tests tests.cpp ${COMPILER_ROOT}/ir_opt/texture_pass.cpp ${IR_SOURCES})
target_include_directories(handle_tracking_tests PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/include ${BOOST_INCLUDE_DIR} ${HOST_DEPS}/range/include)
target_link_libraries(handle_tracking_tests PRIVATE fmt::fmt)
target_compile_options(handle_tracking_tests PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(handle_tracking_tests PRIVATE -fsanitize=undefined)
enable_testing()
add_test(NAME handle_tracking COMMAND handle_tracking_tests)
