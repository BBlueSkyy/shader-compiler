cmake_minimum_required(VERSION 3.16)
project(structured_ssbo_real CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
if(NOT DEPS OR NOT SIRIT_SOURCE)
  message(FATAL_ERROR "Pass DEPS and SIRIT_SOURCE")
endif()
set(FMT_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/fmt" fmt)
add_subdirectory("${SIRIT_SOURCE}" sirit)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(GLOB boost_inc "${DEPS}/boost/libs/*/include")
file(GLOB_RECURSE sources "${root}/frontend/*.cpp" "${root}/ir_opt/*.cpp" "${root}/backend/spirv/*.cpp")
add_executable(real real.cpp ${sources})
target_include_directories(real PRIVATE "${root}/include" ${boost_inc} "${DEPS}/range/include")
target_compile_options(real PRIVATE -include cstddef -fno-strict-aliasing)
target_compile_definitions(real PRIVATE MODERN_SSBO)
target_link_libraries(real PRIVATE fmt::fmt sirit)

# Older pinned range-v3 predates GCC 13's diagnostic becoming an error.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 13)
  target_compile_options(real PRIVATE -Wno-changes-meaning)
endif()
